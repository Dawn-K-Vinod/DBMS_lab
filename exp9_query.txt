CREATE TABLE Bank (
    bankcode    VARCHAR(3)      PRIMARY KEY,
    bankname    VARCHAR(50)     NOT NULL,
    headoffice  VARCHAR(50),
    branches    INT             CHECK (branches > 0)
);

CREATE TABLE Branch (
    branchid    VARCHAR(5)      PRIMARY KEY,
    branchname  VARCHAR(50)     DEFAULT 'New Delhi',
    bankid      VARCHAR(3),
    FOREIGN KEY (bankid) REFERENCES Bank(bankcode)

--1
CREATE VIEW bank_headoffice AS
SELECT *
FROM Bank
WHERE headoffice = 'Ernakulam';
-- Display
SELECT * FROM bank_headoffice;


--2
CREATE VIEW bank_branch AS
SELECT b.*
FROM Bank b
JOIN Branch br ON b.bankcode = br.bankid
WHERE br.branchname = 'Kottayam';
-- Display
SELECT * FROM bank_branch;


--3
CREATE VIEW bank_50branches AS
SELECT *
FROM Bank
WHERE branches > 50;
-- Display
SELECT * FROM bank_50branches;


--4
CREATE VIEW bank_branchname AS
SELECT b.bankname, br.branchname
FROM Bank b
JOIN Branch br ON b.bankcode = br.bankid;
-- Display
SELECT * FROM bank_branchname;


--5
CREATE VIEW bank_outside_ernakulam AS
SELECT *
FROM Bank
WHERE headoffice != 'Ernakulam';
-- Display
SELECT * FROM bank_outside_ernakulam;
